config system global
    set hostname "SANTO_DOMINGO"
    set pre-login-banner enable
    set post-login-banner enable
    set login-banner "Cybershield7 - Unauthorized Access is Prohibited"
end

config system interface
    edit "port2"
        set alias "WAN"
        set role wan
        set mode dhcp
        set allowaccess ping http https ssh
    next

    edit "vlan10"
        set alias "VLAN10"
        set vdom "root"
        set interface "port1"
        set vlanid 10
        set mode static
        set ip 10.136.0.1 255.255.255.128
        set allowaccess ping
    next

    edit "vlan20"
        set alias "VLAN20"
        set vdom "root"
        set interface "port1"
        set vlanid 20
        set mode static
        set ip 10.136.1.129 255.255.255.192
        set allowaccess ping
    next

    edit "vlan30"
        set alias "VLAN30"
        set vdom "root"
        set interface "port1"
        set vlanid 30
        set mode static
        set ip 10.136.1.1 255.255.255.128
        set allowaccess ping
    next

    edit "vlan88"
        set alias "VLAN88"
        set vdom "root"
        set interface "port1"
        set vlanid 88
        set mode static
        set ip 10.136.0.129 255.255.255.128
        set allowaccess ping http https
    next
end

config system dhcp server
    edit 10
        set interface "vlan10"
        set default-gateway 10.136.0.1
        set netmask 255.255.255.128
        set dns-service manual
        set dns-server1 10.128.0.82
        set domain "cybershield7.net"
        config ip-range
            edit 1
                set start-ip 10.136.0.2
                set end-ip   10.136.0.126
            next
        end
    next

    edit 20

config firewall address
    edit "VLAN10_NET"
        set subnet 10.136.0.0/25
    next
    edit "VLAN20_NET"
        set subnet 10.136.1.128/26
    next
    edit "VLAN30_NET"
        set subnet 10.136.1.0/25
    next
    edit "VLAN88_NET"
        set subnet 10.136.0.128/25
    next
end

config firewall addrgrp
    edit "NAT_SD"
        set member "VLAN10_NET" "VLAN20_NET" "VLAN30_NET" "VLAN88_NET"
    next
end

------------------------------------------------------------------------------------------------------------------------------------------------------------------


config firewall policy
    edit 0
        set name "LANs_to_Internet"
        set srcintf "vlan10" "vlan20" "vlan30" "vlan88"
        set dstintf "port2"
        set srcaddr "NAT_SD"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
    next
end

config firewall policy
    edit 0
        set name "InterVLAN"
        set srcintf "vlan10" "vlan20" "vlan30" "vlan88"
        set dstintf "vlan10" "vlan20" "vlan30" "vlan88"
        set srcaddr "NAT_SD"
        set dstaddr "NAT_SD"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat disable
    next
end


config router static
    edit 1
        set dst 0.0.0.0 0.0.0.0
        set gateway 192.168.162.2
        set device "port2"
    next
end
