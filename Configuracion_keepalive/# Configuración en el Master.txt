# Configuración en el Master

vrrp_instance VI_1 {
    state MASTER
    interface ens33                # Cambia por el nombre de la interfaz (ej: ens160, eth0)
    virtual_router_id 51
    priority 150                   # Prioridad mayor que el backup
    advert_int 1
    authentication {
        auth_type PASS
        auth_pass 12345
    }
    virtual_ipaddress {
        10.0.0.101/24           # IP virtual compartida
    }
}


# Configuración en el Backup

vrrp_instance VI_1 {
    state BACKUP
    interface ens33                # Cambia por tu interfaz
    virtual_router_id 51
    priority 100                   # Menor que el master
    advert_int 1
    authentication {
        auth_type PASS
        auth_pass 12345
    }
    virtual_ipaddress {
        192.168.1.100/24
    }
}
